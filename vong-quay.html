async function checkAndSpin() { const phoneInput = document.getElementById('phone'); const btn = document.getElementById('spinBtn'); const resDiv = document.getElementById('result'); const phone = phoneInput.value.trim().replace(/[^0-9]/g, ""); if (phone.length < 10) { resDiv.innerHTML = "
❌ Vui lòng nhập đúng SĐT!
"; return; } btn.disabled = true; btn.innerText = "⌛ ĐANG KIỂM TRA..."; resDiv.innerHTML = ""; try { // 1. Kiểm tra điểm const response = await fetch(`${CONFIG.SCRIPT_URL}?action=get_points&phone=${phone}`); const currentPoints = parseFloat(await response.text()) || 0; if (currentPoints >= CONFIG.SPIN_POINTS) { localStorage.setItem('didi_customer_phone', phone); const totalWeight = GIFT_DATA.reduce((sum, item) => sum + item.weight, 0); let random = Math.random() * totalWeight; let selectedIndex = 0; for (let i = 0; i < GIFT_DATA.length; i++) { if (random < GIFT_DATA[i].weight) { selectedIndex = i; break; } random -= GIFT_DATA[i].weight; } const giftName = GIFT_DATA[selectedIndex].name; const arc = 360 / GIFTS.length; const targetDegree = 360 - (selectedIndex * arc + arc / 2); currentRotation += 1440 + (targetDegree - (currentRotation % 360) + 360) % 360; document.getElementById('wheel').style.transform = `rotate(${currentRotation}deg)`; btn.innerText = "🎡 ĐANG QUAY..."; // 2. Gửi request quay VÀ HỨNG KẾT QUẢ TRẢ VỀ TỪ SCRIPT const spinReq = await fetch(`${CONFIG.SCRIPT_URL}?action=vong_quay&phone=${phone}&gift=${encodeURIComponent(giftName)}&points=${CONFIG.SPIN_POINTS}`); const spinData = await spinReq.json(); // Lấy điểm còn lại trực tiếp từ dữ liệu Script trả về (nếu không có thì mới dự phòng bằng currentPoints - CONFIG.SPIN_POINTS) const remainingPointsReal = (spinData && spinData.remaining_points !== undefined) ? spinData.remaining_points : (currentPoints - CONFIG.SPIN_POINTS); setTimeout(() => { const now = new Date(); const timeStr = now.toLocaleDateString('vi-VN') + " " + now.toLocaleTimeString('vi-VN'); resDiv.innerHTML = `
${giftName.includes('Chúc') ? '🍀' : '🎉'} ${giftName.toUpperCase()}
Thời gian quay: ${timeStr} Số điểm còn lại: ${remainingPointsReal}
`; if (!giftName.includes("Chúc")) { if (typeof confetti === 'function') { confetti({ particleCount: 100, spread: 70, origin: { y: 0.6 } }); } } btn.disabled = false; btn.innerText = "QUAY TIẾP"; }, 4000); } else { resDiv.innerHTML = `
❌ Không đủ điểm! (Hiện có: ${currentPoints})
`; btn.disabled = false; btn.innerText = "QUAY NGAY"; } } catch (e) { resDiv.innerHTML = "
⚠️ Lỗi kết nối!
"; btn.disabled = false; btn.innerText = "THỬ LẠI"; } }
